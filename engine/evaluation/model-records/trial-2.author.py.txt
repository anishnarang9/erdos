def lit(t,v): return {'op':'literal','type':t,'value':str(v) if t=='i64' else v}
def field(src,file,name): return {'op':'field','source':src,'file':file,'field':name}
def rf(name,level=0): return {'op':'record_field','field':name,'level':level}
def binop(op,a,b): return {'op':op,'left':a,'right':b}
def coll(src,file,f='rows'): return {'source':src,'file':file,'field':f}
def allc(c,p): return {'op':'all','collection':c,'where':p}
def anyc(c,p): return {'op':'any','collection':c,'where':p}
def count(c): return {'op':'count','collection':c}
def unique(c,f): return {'op':'unique','collection':c,'field':f}
def lookup(c,kf,key,vf): return {'op':'lookup','collection':c,'keyField':kf,'key':key,'valueField':vf}
def conj(xs):
 if len(xs)==1:return xs[0]
 a=xs[0]
 for x in xs[1:]: a=binop('and',a,x)
 return a
def sourcequote(s): return [{'path':'/rules/request','quote':s}]
def draft(delivs,refs,rules,evidence=None,quotes=''):
 p={'version':1,'deliverables':delivs,'references':refs,'rules':rules}
 if evidence is not None:p['evidence']=evidence
 return {'kind':'draft','plan':p,'examples':[],'sourceQuotes':sourcequote(quotes)}
def file_from(x):
 y={'name':x['name'],'parser':x['parser']}
 if x.get('fields'):y['fields']=x['fields']
 if x.get('collections'):y['collections']=x['collections']
 return y
